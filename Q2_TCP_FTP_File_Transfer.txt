# Two-node TCP file transfer using FTP
set ns [new Simulator]

set nf [open q2.nam w]
$ns namtrace-all $nf
set tf [open q2.tr w]
$ns trace-all $tf

set n0 [$ns node]
set n1 [$ns node]

$ns duplex-link $n0 $n1 1Mb 10ms DropTail
$ns duplex-link-op $n0 $n1 orient right

# TCP sender at node 0
set tcp [new Agent/TCP]
$ns attach-agent $n0 $tcp

# TCP receiver at node 1
set sink [new Agent/TCPSink]
$ns attach-agent $n1 $sink
$ns connect $tcp $sink

# FTP application over TCP
set ftp [new Application/FTP]
$ftp attach-agent $tcp

$ns at 1.0 "$ftp start"
$ns at 5.0 "$ftp stop"

proc finish {} {
    global ns nf tf
    $ns flush-trace
    close $nf
    close $tf
    exec nam q2.nam &
    exit 0
}
$ns at 6.0 "finish"
$ns run
